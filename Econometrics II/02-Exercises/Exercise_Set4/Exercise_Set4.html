6.4 Estimate a pooled logit model, a conditional fixed effect logit model, and a random effects logit model.

# Pooled logit
m1 <- glm( docvis01 ~ age + hhninc + hhkids + educ + married, 
           data=d, family = binomial(link = "logit"))

#Conditional fixed effect logit
library(survival)
m2 <- clogit( as.numeric(d$docvis01) ~ 
                age + hhninc + hhkids + educ + married + 
                strata(id), data = d)

#Random effects logit
library(lme4)
m3 <- glmer( docvis01 ~ 
               age + hhninc + hhkids + educ + married + (1 | id),
             data = d, family = binomial(link = "logit"))
