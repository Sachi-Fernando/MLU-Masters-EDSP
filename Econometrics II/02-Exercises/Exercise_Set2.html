# initializing MCMC
p_sample <- rep(1, 1e4)
plot(p_sample)

p_prior <- function(p) dunif(p, min=0, max=1) # prior on p
p_current <- 0.5  #start value

for (i in 1:1e4) {
  p_sample[i] <- p_current # record current p
  p_proposal <- runif(1, min=0, max=1) #generate proposal (from uniform distribution)
  
  lkhd_current <- dbinom(8, 15, p_current) #8 water tosses out of 15 (data observed)
  lkhd_proposal <- dbinom(8, 15, p_proposal)
  
  prob_accept <- (lkhd_proposal *p_prior(p_proposal) ) / ( lkhd_current * p_prior(p_current) )
  p_current <- ifelse( runif(1) < prob_accept, p_proposal, p_current)
}
# Drop warm-up:
burn_in <- 1000
samples_post <- p_sample[-(1:burn_in)]

#MCMC-Samples from posterior
plot(samples_post, type="l", col="royalblue4")

dens(samples_post, col="royalblue4")
